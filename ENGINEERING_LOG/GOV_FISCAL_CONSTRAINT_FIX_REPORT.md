function taxOf(city) {
   const siRate  = COSTS.tax.socialInsurance.employee.total;
   const vatRate = COSTS.tax.consumptionTax.effectiveRate;
   let pit = 0, si = 0, vat = 0, nTaxpayer = 0;

}
